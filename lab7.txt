# =========================
# models.py
# =========================
from django.db import models

class Student(models.Model):
    name = models.CharField(max_length=100)
    usn = models.CharField(max_length=20)
    semester = models.IntegerField()
    fee_paid = models.BooleanField()


# =========================
# forms.py
# =========================
from django import forms
from .models import Student

class StudentForm(forms.ModelForm):
    class Meta:
        model = Student
        fields = ['name', 'usn', 'semester', 'fee_paid']


# =========================
# views.py
# =========================
from django.shortcuts import render, redirect
from .models import Student
from .forms import StudentForm

def home(request):
    if request.method == 'POST':
        form = StudentForm(request.POST)
        if form.is_valid():
            form.save()
            return redirect('list')
    else:
        form = StudentForm()
    return render(request, 'home.html', {'form': form})


def student_list(request):
    students = Student.objects.all()
    return render(request, 'list.html', {'students': students})


def delete_unpaid(request):
    if request.method == 'POST':
        Student.objects.filter(fee_paid=False).delete()
    return redirect('list')


# =========================
# studentapp/urls.py
# =========================
from django.urls import path
from . import views

urlpatterns = [
    path('', views.home, name='home'),
    path('list/', views.student_list, name='list'),
    path('delete/', views.delete_unpaid, name='delete'),
]
